# Donor Incentive & Resource Commitment Engine - Implementation Guide

## Overview

This guide provides a complete implementation blueprint for the Donor Incentive & Resource Commitment Engine—a sophisticated system that resolves the core loophole in Tech-Equity Bridge by creating formal incentive mechanisms, quality enforcement, and commitment tracking for tech donors.

## Problem Statement

The current platform assumes large tech companies have genuinely idle GPU/AI capacity they're willing to donate. This assumption is fragile because:

- **Cloud auto-scaling**: Providers auto-scale down idle capacity, so it doesn't stay "idle"
- **Throttled resources**: Donors may only contribute low-priority, throttled resources giving non-profits second-class performance
- **No incentive mechanism**: No clear answer to WHY a tech giant would donate instead of selling capacity

## Solution: Three-Tier Incentive Framework

### Tier 1: "Impact Ally" ($500/month equivalent)
**Benefits:**
- Public donor badge on platform homepage
- Monthly auto-generated CSR impact report (PDF) showing usage + outcomes
- Access to anonymized non-profit impact stories for internal comms

**Requirements:**
- Minimum 500 GPU-hours OR 50,000 API calls OR 1,000 agent-hours per month
- Response latency < 5s p95
- Uptime > 95%

### Tier 2: "Equity Champion" ($2,000–5,000/month equivalent)
**Benefits:**
- All Tier 1 benefits
- Featured placement in the Agent Marketplace with donor branding
- Quarterly joint press release template auto-generated by platform
- Named attribution on non-profit success stories

**Requirements:**
- Minimum 2,000 GPU-hours OR 200,000 API calls OR 5,000 agent-hours per month
- Response latency < 3s p95
- Uptime > 99%
- Quality score ≥ 4.0/5.0

### Tier 3: "Founding Partner" (enterprise-level, long-term)
**Benefits:**
- All Tier 2 benefits
- Co-branded impact microsite (auto-generated, hosted on platform subdomain)
- Seat on the Platform Advisory Council
- Priority API integration support from Tech-Equity Bridge team
- Verified "ESG Contribution Certificate" (blockchain-anchored or signed PDF)

**Requirements:**
- Minimum 10,000 GPU-hours OR 1,000,000 API calls OR 50,000 agent-hours per month
- Response latency < 2s p95
- Uptime > 99.5%
- Quality score ≥ 4.5/5.0
- 12-month minimum commitment

## Database Schema

### 1. Donor Incentive Tiers Table
```sql
CREATE TABLE donorIncentiveTiers (
  id INT PRIMARY KEY AUTO_INCREMENT,
  donorId INT NOT NULL UNIQUE,
  tier ENUM('impact_ally', 'equity_champion', 'founding_partner') NOT NULL,
  monthlyPledgeAmount DECIMAL(12,2) NOT NULL,
  pledgeUnit ENUM('gpu_hours', 'api_calls', 'agent_hours', 'compute_units') NOT NULL,
  verificationStatus ENUM('pending', 'verified', 'rejected') DEFAULT 'pending',
  badgePublic BOOLEAN DEFAULT TRUE,
  csrReportsEnabled BOOLEAN DEFAULT TRUE,
  createdAt TIMESTAMP DEFAULT NOW(),
  updatedAt TIMESTAMP DEFAULT NOW() ON UPDATE CURRENT_TIMESTAMP
);
```

### 2. Resource Pledges Table
```sql
CREATE TABLE resourcePledges (
  id INT PRIMARY KEY AUTO_INCREMENT,
```

```typescript
  startDate: Date;

// Fulfillment Tracking
export async function logPledgeFulfillment(pledgeId: number, month: string, data: {
  pledgedAmount: number;
  deliveredAmount: number;
}) {
  const db = await getDb();
  if (!db) throw new Error("Database not available");
  
  const fulfillmentPercentage = (data.deliveredAmount / data.pledgedAmount) * 100;
  const flagged = fulfillmentPercentage < 80;
  
  const result = await db.insert(pledgeFulfillmentLog).values({
    pledgeId,
    month,
    pledgedAmount: data.pledgedAmount,
    deliveredAmount: data.deliveredAmount,
    fulfillmentPercentage: new Decimal(fulfillmentPercentage),
    flagged,
    flagReason: flagged ? 'Under-delivery detected' : undefined,
  });
  
  return result;
}

// Quality Benchmark Testing
export async function recordQualityBenchmark(resourceId: number, data: {
  resourceType: string;
  latencyP95Ms?: number;
  uptimePercentage?: number;
  tokenLimit?: number;
  throughputBenchmark?: string;
  jobCompletionSlaHours?: number;
}) {
  const db = await getDb();
  if (!db) throw new Error("Database not available");
  
  // Calculate quality score based on SLA compliance
  let qualityScore = 5.0;
  
  if (data.latencyP95Ms && data.latencyP95Ms > 3000) qualityScore -= 1;
  if (data.uptimePercentage && data.uptimePercentage < 99) qualityScore -= 1;
  if (data.tokenLimit && data.tokenLimit < 4000) qualityScore -= 1;
  
  const benchmarkPassed = qualityScore >= 3.0;
  
  const result = await db.insert(resourceQualityBenchmarks).values({
    resourceId,
    resourceType: data.resourceType,
    ...data,
    qualityScore: new Decimal(qualityScore),
    benchmarkPassed,
    testedAt: new Date(),
  });
  
  return result;
}

// CSR Report Generation
export async function generateCsrReport(donorId: number, month: string) {
  const db = await getDb();
  if (!db) throw new Error("Database not available");
  
  // Aggregate metrics for the month
  const pledges = await db.query.resourcePledges.findMany({
    where: eq(resourcePledges.donorId, donorId),
  });
  
  const fulfillment = await db.query.pledgeFulfillmentLog.findMany({
    where: and(
      eq(pledgeFulfillmentLog.month, month),
      inArray(pledgeFulfillmentLog.pledgeId, pledges.map(p => p.id))
    ),
  });
  
  const totalHours = fulfillment.reduce((sum, f) => sum + Number(f.deliveredAmount), 0);
  
  // Create CSR report
  const result = await db.insert(csrReports).values({
    donorId,
    month,
    hoursContributed: new Decimal(totalHours),
    organizationsHelped: 0, // Aggregate from requests
    peopleImpacted: 0, // Aggregate from impact metrics
    griAligned: true,
  });
  
  return result;
}

// Incentive Event Logging
export async function logIncentiveEvent(donorId: number, eventType: string, details: Record<string, unknown>) {
  const db = await getDb();
  if (!db) throw new Error("Database not available");
  
  const result = await db.insert(donorIncentiveEvents).values({
    donorId,
    eventType: eventType as any,
    details,
  });
  
  return result;
}
```

```typescript
// Donor Incentive Router
export const donorIncentiveRouter = router({
  // Create pledge
  createPledge: protectedProcedure
    .input(z.object({
      resourceType: z.enum(['ai_agent', 'gpu_compute', 'data_processing', 'software_tool']),
      quantity: z.number().positive(),
      unit: z.string(),
      availabilityWindows: z.array(z.object({
        day: z.string(),
        startTime: z.string(),
        endTime: z.string(),
      })),
      startDate: z.date(),
      endDate: z.date().optional(),
    }))
    .mutation(async ({ ctx, input }) => {
      if (ctx.user.role !== 'donor') throw new TRPCError({ code: 'FORBIDDEN' });
      
      const donor = await getDonorProfile(ctx.user.id);
      if (!donor) throw new TRPCError({ code: 'NOT_FOUND' });
      
      const pledge = await createResourcePledge(donor.id, input);
      
      await logIncentiveEvent(donor.id, 'pledge_created', {
        pledgeId: pledge.insertId,
        resourceType: input.resourceType,
        quantity: input.quantity,
      });
      
      return pledge;
    }),

  // Get pledges
  getPledges: protectedProcedure.query(async ({ ctx }) => {
    if (ctx.user.role !== 'donor') throw new TRPCError({ code: 'FORBIDDEN' });
    
    const donor = await getDonorProfile(ctx.user.id);
    if (!donor) throw new TRPCError({ code: 'NOT_FOUND' });
    
    return await getResourcePledges(donor.id);
  }),

  // Get fulfillment status
  getFulfillmentStatus: protectedProcedure
    .input(z.object({ pledgeId: z.number() }))
    .query(async ({ ctx, input }) => {
      const pledge = await getResourcePledge(input.pledgeId);
      if (!pledge) throw new TRPCError({ code: 'NOT_FOUND' });
      
      if (pledge.donorId !== ctx.user.id && ctx.user.role !== 'admin') {
        throw new TRPCError({ code: 'FORBIDDEN' });
      }
      
      return await getPledgeFulfillmentLog(input.pledgeId);
    }),

  // Get quality score
  getQualityScore: protectedProcedure
    .input(z.object({ resourceId: z.number() }))
    .query(async ({ input }) => {
      return await getResourceQualityBenchmark(input.resourceId);
    }),

  // Generate CSR report
  generateCsrReport: protectedProcedure
    .input(z.object({ month: z.string() }))
    .mutation(async ({ ctx, input }) => {
      if (ctx.user.role !== 'donor') throw new TRPCError({ code: 'FORBIDDEN' });
      
      const donor = await getDonorProfile(ctx.user.id);
      if (!donor) throw new TRPCError({ code: 'NOT_FOUND' });
      
      const report = await generateCsrReport(donor.id, input.month);
      
      // Generate PDF (see PDF generation section)
      const pdfUrl = await generateCsrPdf(donor.id, input.month);
      
      // Update report with URL
      await updateCsrReportUrl(report.insertId, pdfUrl);
      
      return { reportId: report.insertId, pdfUrl };
    }),

  // Get impact wall
  getImpactWall: publicProcedure
    .input(z.object({ slug: z.string() }))
    .query(async ({ input }) => {
      return await getDonorImpactWallBySlug(input.slug);
    }),

  // Get all impact walls (for public page)
  getAllImpactWalls: publicProcedure.query(async () => {
    return await getAllDonorImpactWalls();
  }),
});
```

```typescript
import { Queue, Worker } from 'bullmq';

const fulfillmentQueue = new Queue('fulfillment-tracking', {
  connection: {
    host: process.env.REDIS_HOST || 'localhost',
    port: parseInt(process.env.REDIS_PORT || '6379'),
  },
});

// Job: Track monthly fulfillment
fulfillmentQueue.add(
  'monthly-fulfillment',
  { month: '2026-01' },
  {
    repeat: {
      pattern: '0 0 1 * *', // First day of each month
    },
  }
);

// Worker: Process fulfillment tracking
new Worker('fulfillment-tracking', async (job) => {
  if (job.name === 'monthly-fulfillment') {
    const { month } = job.data;
    
    // Get all active pledges
    const pledges = await getAllActivePledges();
    
    for (const pledge of pledges) {
      // Calculate delivered amount (from resource usage logs)
      const delivered = await calculateDeliveredAmount(pledge.id, month);
      const pledged = Number(pledge.quantity);
      
      // Log fulfillment
      await logPledgeFulfillment(pledge.id, month, {
        pledgedAmount: pledged,
        deliveredAmount: delivered,
      });
      
      // Check for under-delivery
      const fulfillmentPct = (delivered / pledged) * 100;
      if (fulfillmentPct < 80) {
        // Trigger tier downgrade
        await evaluateTierDowngrade(pledge.donorId, month);
      }
    }
  }
}, {
  connection: {
    host: process.env.REDIS_HOST || 'localhost',
    port: parseInt(process.env.REDIS_PORT || '6379'),
  },
});

// jobs/tierEvaluation.ts
// Job: Evaluate tier downgrades
tierEvaluationQueue.add(
  'monthly-tier-evaluation',
  {},
  {
    repeat: {
      pattern: '0 1 1 * *', // 1 AM on first day of month
    },
  }
);

// Worker: Process tier evaluation
new Worker('tier-evaluation', async (job) => {
  const donors = await getAllDonors();
  
  for (const donor of donors) {
    const tier = await getDonorIncentiveTier(donor.id);
    if (!tier) continue;
    
    // Get last month's fulfillment
    const lastMonth = getLastMonth();
    const fulfillment = await getPledgeFulfillmentLog(tier.id, lastMonth);
    
    if (fulfillment.fulfillmentPercentage < 80) {
      // Check grace period
      const graceEvents = await getGracePeriodEvents(donor.id);
      
      if (graceEvents.length < 1) {
        // Apply grace period
        await logIncentiveEvent(donor.id, 'grace_period_applied', {
          tier: tier.tier,
          fulfillmentPct: fulfillment.fulfillmentPercentage,
          expiresAt: addDays(new Date(), 7),
        });
      } else {
        // Downgrade tier
        const newTier = downgradeTier(tier.tier);
        await updateDonorIncentiveTier(donor.id, { tier: newTier });
        
        await logIncentiveEvent(donor.id, 'tier_downgrade', {
          from: tier.tier,
          to: newTier,
          reason: 'Under-delivery',
        });
        
        // Send email notification
        await sendTierDowngradeEmail(donor.email, newTier);
      }
    }
  }
}, {
  connection: { host: process.env.REDIS_HOST, port: parseInt(process.env.REDIS_PORT) },
});

// jobs/csrReportGenerator.ts
// Job: Generate CSR reports
csrReportQueue.add(
  'monthly-csr-generation',
  {},
  {
    repeat: {
      pattern: '0 2 1 * *', // 2 AM on first day of month
    },
  }
);

// Worker: Generate CSR reports
new Worker('csr-generation', async (job) => {
  const donors = await getAllDonors();
  
  for (const donor of donors) {
    const tier = await getDonorIncentiveTier(donor.id);
    if (!tier || !tier.csrReportsEnabled) continue;
    
    const lastMonth = getLastMonth();
    const report = await generateCsrReport(donor.id, lastMonth);
    
    // Generate PDF
    const pdfUrl = await generateCsrPdf(donor.id, lastMonth);
    await updateCsrReportUrl(report.id, pdfUrl);
    
    // Send email
    await sendCsrReportEmail(donor.email, pdfUrl, tier.tier);
  }
}, {
  connection: { host: process.env.REDIS_HOST, port: parseInt(process.env.REDIS_PORT) },
});
```

## Quality Benchmark Testing

```typescript
// server/services/benchmarkRunner.ts
export async function runBenchmarkTest(resourceId: number, resourceType: string) {
  const resource = await getResource(resourceId);
  
  let latencyP95Ms: number | undefined;
  let uptimePercentage: number | undefined;
  let tokenLimit: number | undefined;
  
  if (resourceType === 'ai_agent') {
    // Test latency with 100 requests
    const latencies = [];
    for (let i = 0; i < 100; i++) {
      const start = Date.now();
      await callResource(resource.endpoint);
      latencies.push(Date.now() - start);
    }
    latencies.sort((a, b) => a - b);
    latencyP95Ms = latencies[Math.floor(latencies.length * 0.95)];
    
    // Test uptime with 10 requests
    let successCount = 0;
    for (let i = 0; i < 10; i++) {
      try {
        await callResource(resource.endpoint);
        successCount++;
      } catch (e) {
        // Failed request
      }
    }
    uptimePercentage = (successCount / 10) * 100;
    
    // Check token limit
    tokenLimit = resource.tokenLimit || 4000;
  }
  
  // Record benchmark
  await recordQualityBenchmark(resourceId, {
    resourceType,
    latencyP95Ms,
    uptimePercentage,
    tokenLimit,
  });
}
```
